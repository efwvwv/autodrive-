Парсер каталога AutoDrive

Превращает выгрузку с сайта-источника в готовый data.js: фото и описание берутся из одной карточки, поэтому совпадают.

Шаг 1. Выгрузка с сайта

Откройте каталог источника, нажимайте «Показать ещё» до конца списка, откройте консоль (F12) и выполните код. Результат окажется в буфере обмена.

(() => {
  const map = new Map();
  document.querySelectorAll('img').forEach(img => {
    if (!img.src.includes('/uploads/cars/')) return;
    let el = img;
    for (let i = 0; i < 8 && el.parentElement; i++) {
      el = el.parentElement;
      if (el.innerText && el.innerText.trim().length > 25) break;
    }
    const text = el.innerText.trim().replace(/\n+/g, ' | ');
    if (!map.has(text)) map.set(text, []);
    const a = map.get(text);
    if (!a.includes(img.src)) a.push(img.src);
  });
  const out = [...map].map(([text, photos]) => ({ text, photos }));
  copy(JSON.stringify(out));
  console.log('Карточек:', out.length);
})();

Шаг 2. Вставьте выгрузку

Принимает и короткий формат {"t":…,"p":[…]}, и полный {"text":…,"photos":[…]}. Можно вставить несколько частей подряд.
